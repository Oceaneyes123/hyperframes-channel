async(page)=>{
 const ids={'rolling-value':'$120.00','value-transition':'10,000',percent:'32.50%'},results=[];
 const svg=page.locator('svg[data-motion-frame]'),slider=page.getByRole('slider',{name:'Inspect frame'});
 const seek=async frame=>{await slider.press('Home');for(let i=0;i<frame;i++)await slider.press('ArrowRight');await page.waitForFunction(f=>document.querySelector('svg[data-motion-frame]')?.getAttribute('data-motion-frame')===String(f),frame);};
 for(const [id,expected] of Object.entries(ids)){
  await page.getByRole('combobox',{name:'Capability'}).selectOption(id);await seek(36);const before=await svg.evaluate(el=>el.outerHTML);await svg.screenshot({path:'.planning/2026-10-25-advanced-motion-plan/'+id+'-before.png'});
  await page.getByRole('button',{name:'End state',exact:true}).click();await page.waitForFunction(()=>document.querySelector('svg[data-motion-frame]')?.getAttribute('data-motion-frame')==='119');
  const visible=await svg.locator('[data-part="digits"]').evaluate(part=>{const clip=document.getElementById(part.getAttribute('clip-path').match(/#([^)]*)/)[1]).querySelector('rect'),height=Number(clip.getAttribute('height'));return [...part.querySelectorAll('path[data-numeric-character]')].filter(text=>{if(Number(getComputedStyle(text).opacity)<.9)return false;const box=text.getBBox(),p=new DOMPoint(box.x+box.width/2,box.y+box.height/2).matrixTransform(text.getCTM()).matrixTransform(part.getCTM().inverse());return p.y>0&&p.y<height;}).map(text=>text.getAttribute('data-numeric-character')).join('').trim();});
  if(visible!==expected)throw Error(id+' visible value '+JSON.stringify(visible)+' expected '+expected);
  await page.screenshot({path:'.planning/2026-10-25-advanced-motion-plan/'+id+'-end.png',fullPage:true});await seek(0);await seek(36);if(await svg.evaluate(el=>el.outerHTML)!==before)throw Error(id+' mounted backward mismatch');await svg.screenshot({path:'.planning/2026-10-25-advanced-motion-plan/'+id+'-after.png'});
  results.push({id,visible,mountedBackward:'exact'});
 }
 await page.getByRole('slider',{name:/Roll duration/}).press('Home');await page.getByRole('button',{name:'End state',exact:true}).click();
 await page.setViewportSize({width:390,height:844});await page.screenshot({path:'.planning/2026-10-25-advanced-motion-plan/numeric-mobile.png',fullPage:true});return results;
}
